完整 doEncrypt() 详细还原版：

bool doEncrypt(
    long timestamp_ms,
    const double& lon,
    const double& lat,
    const double& height,
    double& out_lon,
    double& out_lat
) {
    // ---------- 时间处理 ----------
    time_t sec = timestamp_ms / 1000;
    tm* utc = gmtime(&sec);

    int gps_week = 0;
    int gps_sow = 0;

    UTC2GPS(
        utc->tm_year + 1900,
        utc->tm_mon + 1,
        utc->tm_mday,
        utc->tm_hour,
        utc->tm_min,
        utc->tm_sec,
        gps_week,
        gps_sow
    );

    // ---------- 基础合法性检查 ----------
    if ((int)height > 5000) {
        LOG(ERROR) << "gnss lon lat encrypt failed";
        return false;
    }

    // 0x96c = 2412, 0x1c = 28，因此允许范围是 2412 ~ 2440。
    if (gps_week < 2412 || gps_week > 2440) {
        LOG(ERROR) << "gnss lon lat encrypt failed";
        return false;
    }

    // 反编译出的经纬度范围常量。当前 11 个 bin 中该范围一致。
    if (lon < 115.73278 || lon > 117.85876 ||
        lat < 38.51527  || lat > 40.41024) {
        LOG(ERROR) << "gnss lon lat encrypt failed";
        return false;
    }

    // ---------- 经纬度量化 ----------
    double q_lon = floor(lon * 3686400.0) / 3686400.0;
    double q_lat = floor(lat * 3686400.0) / 3686400.0;

    // ---------- 首帧初始化 ----------
    if (g_first_encrypt) {
        g_state.gps_sow = gps_sow;
        g_state.lon0 = q_lon;
        g_state.lon1 = q_lon;
        g_state.lat0 = q_lat;
        g_state.lat1 = q_lat;
        g_state.counter = 3.0;

        out_lon = q_lon;
        out_lat = q_lat;

        g_first_encrypt = false;
        return true;
    }

    // ---------- 跳变检测 ----------
    if (JudgeSD(gps_sow, q_lon, q_lat) == -1) {
        LOG(ERROR) << "gnss lon lat encrypt failed";
        return false;
    }

    // ---------- 坐标规整 ----------
    double base_lon = comLonValue(q_lon);
    double base_lat = comLonValue(q_lat);

    // GCJ 类算法通常使用：x = lon - 105, y = lat - 35。
    double x = base_lon - 105.0;
    double y = base_lat - 35.0;

    // ---------- 平面扰动 ----------
    double d_lon_meter = EncrpytLonLatA(x, y);
    double d_lat_meter = EncrpytLonLatB(x, y);

    // ---------- 高程/区域修正 ----------
    double elev = Elev_Inter(base_lon, base_lat);

    d_lon_meter += elev * 0.357;
    d_lat_meter += elev * 0.357;

    // ---------- 椭球换算 ----------
    double rad_lat = base_lat * DEG_TO_RAD;
    double sin_lat = encrpytTL(rad_lat);

    double magic = 1.0 - EE * sin_lat * sin_lat;
    double sqrt_magic = sqrt(magic);

    double dlon =
        d_lon_meter * 180.0 /
        ((A / sqrt_magic) * cos(rad_lat) * PI);

    double dlat =
        d_lat_meter * 180.0 /
        ((A * (1.0 - EE)) / (magic * sqrt_magic) * PI);

    // ---------- 输出量化 ----------
    out_lon = floor((base_lon + dlon) * 3686400.0) / 3686400.0;
    out_lat = floor((base_lat + dlat) * 3686400.0) / 3686400.0;

    return true;
}

子函数算法逻辑：

1. UTC2GPS(...)
- 从 1980 年开始累计天数。
- 按闰年规则计算每年天数。
- 累加月份和日期。
- days -= 6，修正 GPS epoch。
- gps_week = days / 7。
- gps_sow = day_of_week * 86400 + hour * 3600 + minute * 60 + second + 18。

2. EncrpytLonLatA(x, y)
作用：计算经度方向平面扰动，结构类似 GCJ transformLon。

double EncrpytLonLatA(double x, double y) {
    double ret = 300.0 + x + 2.0 * y + 0.1 * x * x + 0.1 * x * y + 0.1 * sqrt(abs(x));
    ret += (20.0 * sin(6.0 * x * PI) + 20.0 * sin(2.0 * x * PI)) * 2.0 / 3.0;
    ret += (20.0 * sin(x * PI) + 40.0 * sin(x / 3.0 * PI)) * 2.0 / 3.0;
    ret += (150.0 * sin(x / 12.0 * PI) + 300.0 * sin(x / 30.0 * PI)) * 2.0 / 3.0;
    return ret;
}

3. EncrpytLonLatB(x, y)
作用：计算纬度方向平面扰动，结构类似 GCJ transformLat。

double EncrpytLonLatB(double x, double y) {
    double ret = -100.0 + 2.0 * x + 3.0 * y + 0.2 * y * y + 0.1 * x * y + 0.2 * sqrt(abs(x));
    ret += (20.0 * sin(6.0 * x * PI) + 20.0 * sin(2.0 * x * PI)) * 2.0 / 3.0;
    ret += (20.0 * sin(y * PI) + 40.0 * sin(y / 3.0 * PI)) * 2.0 / 3.0;
    ret += (160.0 * sin(y / 12.0 * PI) + 320.0 * sin(y * PI / 30.0)) * 2.0 / 3.0;
    return ret;
}

4. encrpytTL(double x)
作用：自定义 sin 近似函数。先规约到周期，再用 Taylor 多项式近似：

sin(x) ~= x - x^3/6 + x^5/120 - x^7/5040 + x^9/362880 - x^11/39916800 + x^13/6227020800 - x^15/1307674368000

5. Elev_Inter(lon, lat)
作用：区域高程/修正表双线性插值。
常量：LON_TABLE_MIN=72.004, LON_TABLE_MAX=137.8347, LAT_TABLE_MIN=0.8293, LAT_TABLE_MAX=55.8271, GRID_STEP=0.8333, GRID_AREA=0.69438889。
输出会乘以 0.357 加入经纬方向扰动。

6. JudgeSD(...)
作用：轨迹连续性 / GNSS 跳变检查。
- 维护上一时刻 gps_sow/lon/lat。
- 若 dt <= 0，则更新状态并通过。
- 若 dt <= 120，则认为正常并通过。
- 若间隔较大，则用距离 / 时间差构造 speed_like。
- speed_like > 3185.0 时返回 -1，触发加密失败。

7. EncryptHX / EncryptHY
作用：把平面偏移量换算成经纬度角度偏移。

EncryptHX：
    dx * 180 / ((A / sqrt_magic) * cos(rad_lat) * PI)

EncryptHY：
    dy * 180 / ((A * (1 - EE)) / (magic * sqrt_magic) * PI)

关键结论：
- doEncrypt 不是密码学加密，而是 GNSS 经纬度坐标加密 / 偏移算法。
- 该算法使用 GCJ 类 transformLon / transformLat 公式。
- 椭球参数为 a=6378245.0, ee=0.00669342。
- 11 个 bin 的 doEncrypt 代码段和 helper 代码段完全一致；差异不在该算法实现。
